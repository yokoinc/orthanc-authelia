# Dependencies needed to run the test suite, absent from the production
# image. Versions are pinned: without that, pip installs the latest published
# ones and the suite stops being reproducible from one machine to the next.
#
# Run the tests without installing Python on the host:
#
#   docker run --rm --entrypoint sh \
#     -e AUTH_USERNAME=x -e AUTH_PASSWORD=xxxxxxxxxxxx \
#     -e ORTHANC_ADMIN_USER=x -e ORTHANC_ADMIN_PASS=y \
#     -v "$PWD:/depot" -w /depot/services/auth-service/sources \
#     registry.yokoinc.ovh/orthanc-auth-service:1.0.16 \
#     -c 'pip install -q -r requirements-dev.txt && python -m pytest tests/ -q'
#
# Run it from the REPOSITORY ROOT, mounting that root: a few tests reread
# files shipped there. Mounting only sources/ makes them skip themselves --
# that is how a suite green locally broke CI.
#
# The four environment variables are required when the auth_service module
# loads; their values do not matter for the tests.

pytest==9.1.1
pytest-asyncio==1.4.0
fakeredis==2.37.0
respx==0.23.1
